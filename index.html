設定教學
1
確認試算表有四個工作表
前往你的 Google 試算表，確認底部有四個分頁：trips itinerary budget checklist
2
開啟 Apps Script
試算表上方選單 →「擴充功能」→「Apps Script」
把編輯器裡原有的程式碼全部清掉
3
貼上程式碼
貼入下方的程式碼後，按「儲存」（Ctrl+S）
4
部署為網路應用程式
右上角「部署」→「新增部署作業」
▸ 類型：網路應用程式
▸ 執行身分：我（你的帳號）
▸ 誰可以存取：所有人
→ 點「部署」→ 複製產生的網址
5
貼入網址並測試
把複製的網址貼到上方欄位 → 點「儲存網址」→ 點「測試連線」
看到「✅ 連線成功」就完成了！

function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet() .getSheetByName(e.parameter.sheet || 'itinerary'); var data = sheet.getDataRange().getValues(); return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var payload = JSON.parse(e.postData.contents); var sheet = SpreadsheetApp.getActiveSpreadsheet() .getSheetByName(payload.sheet || 'itinerary'); if (payload.action === 'write') { if (sheet.getLastRow() > 1) sheet.getRange(2,1,sheet.getLastRow()-1, sheet.getLastColumn()).clearContent(); payload.rows.forEach(function(r){ sheet.appendRow(r); }); } else if (payload.action === 'append') { sheet.appendRow(payload.row); } return ContentService .createTextOutput(JSON.stringify({ok:true})) .setMimeType(ContentService.MimeType.JSON); }
